// زيادة دقة التقريب للمنحنيات
private _approximateBezierLength(p0: {x:number, y:number}, p1: {x:number, y:number}, p2: {x:number, y:number}, p3: {x:number, y:number}): number {
    // استخدام تقنية تقسيم المنحنى لتحسين الدقة
    const samplePoints = 20;
    let length = 0;
    let prevPoint = p0;
    
    for (let i = 1; i <= samplePoints; i++) {
        const t = i / samplePoints;
        const point = this._calculateBezierPoint(p0, p1, p2, p3, t);
        length += Math.sqrt(
            Math.pow(point.x - prevPoint.x, 2) + 
            Math.pow(point.y - prevPoint.y, 2)
        );
        prevPoint = point;
    }
    
    return length;
}

private _calculateBezierPoint(p0: any, p1: any, p2: any, p3: any, t: number) {
    const u = 1 - t;
    return {
        x: u*u*u*p0.x + 3*u*u*t*p1.x + 3*u*t*t*p2.x + t*t*t*p3.x,
        y: u*u*u*p0.y + 3*u*u*t*p1.y + 3*u*t*t*p2.y + t*t*t*p3.y
    };
}
