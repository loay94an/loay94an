// ---------------------------------------------------------------
//  BASIC TYPES
// ---------------------------------------------------------------
export interface Point { x: number; y: number; }
export interface Polyline extends Array<Point> {}
export interface SvgPath { d?: string; class?: string; label?: string; x?: number; y?: number; }

// ---------------------------------------------------------------
//  PATH BUILDER (clean SVG generator)
// ---------------------------------------------------------------
class PathBuilder {
  private d: string[] = [];

  moveTo(x: number, y: number) { this.d.push(`M ${x} ${y}`); return this; }
  lineTo(x: number, y: number) { this.d.push(`L ${x} ${y}`); return this; }
  cubic(x1:number,y1:number,x2:number,y2:number,x:number,y:number) {
    this.d.push(`C ${x1} ${y1}, ${x2} ${y2}, ${x} ${y}`); return this;
  }
  quad(x1:number,y1:number,x:number,y:number) {
    this.d.push(`Q ${x1} ${y1}, ${x} ${y}`); return this;
  }
  arc(rx:number,ry:number,rot:number,laf:number,sf:number,x:number,y:number) {
    this.d.push(`A ${rx} ${ry} ${rot} ${laf} ${sf} ${x} ${y}`); return this;
  }
  close() { this.d.push('Z'); return this; }
  build() { return this.d.join(' '); }
}

// ---------------------------------------------------------------
//  ADAPTIVE BEZIER SAMPLING (de Casteljau)
// ---------------------------------------------------------------
function adaptiveSampleBezier(
  p0:Point, p1:Point, p2:Point, p3:Point,
  maxSeg = 0.5, maxDepth = 14
): Point[] {

  const out: Point[] = [];
  const toArr = (p:Point) => [p.x, p.y] as [number,number];

  const bez = (a:[number,number],b:[number,number],c:[number,number],d:[number,number],t:number) => {
    const u = 1 - t;
    return {
      x: u*u*u*a[0] + 3*u*u*t*b[0] + 3*u*t*t*c[0] + t*t*t*d[0],
      y: u*u*u*a[1] + 3*u*u*t*b[1] + 3*u*t*t*c[1] + t*t*t*d[1]
    };
  };

  const subdiv = (a:[number,number],b:[number,number],c:[number,number],d:[number,number],depth:number) => {
    const mid = bez(a,b,c,d,0.5);
    const lm = [(a[0]+d[0])/2,(a[1]+d[1])/2];
    const dx = mid.x - lm[0], dy = mid.y - lm[1];

    if (Math.hypot(dx,dy) > maxSeg && depth < maxDepth) {
      const a1:[number,number] = [(a[0]+b[0])/2,(a[1]+b[1])/2];
      const b1:[number,number] = [(b[0]+c[0])/2,(b[1]+c[1])/2];
      const c1:[number,number] = [(c[0]+d[0])/2,(c[1]+d[1])/2];
      const a2:[number,number] = [(a1[0]+b1[0])/2,(a1[1]+b1[1])/2];
      const b2:[number,number] = [(b1[0]+c1[0])/2,(b1[1]+c1[1])/2];
      const m:[number,number] = [(a2[0]+b2[0])/2,(a2[1]+b2[1])/2];

      subdiv(a,a1,a2,m,depth+1);
      subdiv(m,b2,c1,d,depth+1);
    } else {
      out.push({ x:a[0], y:a[1] });
      out.push({ x:d[0], y:d[1] });
    }
  };

  subdiv(toArr(p0),toArr(p1),toArr(p2),toArr(p3),0);

  // remove duplicates
  const res: Point[] = [];
  for (const pt of out)
    if (!res.length || Math.hypot(res[res.length-1].x-pt.x,res[res.length-1].y-pt.y)>1e-9)
      res.push(pt);

  return res;
}

// ---------------------------------------------------------------
//  BEZIER LENGTH (via adaptive sampling)
// ---------------------------------------------------------------
function bezierLength(p0:Point,p1:Point,p2:Point,p3:Point): number {
  const pts = adaptiveSampleBezier(p0,p1,p2,p3,0.5,14);
  let L = 0;
  for (let i=1;i<pts.length;i++)
    L += Math.hypot(pts[i].x-pts[i-1].x, pts[i].y-pts[i-1].y);
  return L;
}

// ---------------------------------------------------------------
//  SVG PATH PARSER (supports M,L,C,Q,A,Z)
// ---------------------------------------------------------------
function parsePath(d: string) {
  const out: { cmd:string; args:number[] }[] = [];
  const re = /([a-zA-Z])([^a-zA-Z]*)/g;
  let m;
  while ((m = re.exec(d)) !== null) {
    const cmd = m[1].toUpperCase();
    const nums = m[2].trim()
      ? m[2].trim().split(/[\s,]+/).map(Number).filter(n=>!isNaN(n))
      : [];
    out.push({ cmd, args: nums });
  }
  return out;
}

// ---------------------------------------------------------------
//  DENSIFY PATH (convert curves → polyline)
// ---------------------------------------------------------------
function densifyPath(d: string, maxSeg = 0.5): Polyline {
  const tokens = parsePath(d);
  const pts: Polyline = [];
  let cur: Point = { x:0, y:0 };

  for (const t of tokens) {
    const cmd = t.cmd;
    const a = t.args;

    if (cmd === 'M') {
      cur = { x:a[0], y:a[1] };
      pts.push({ ...cur });
    }

    else if (cmd === 'L') {
      cur = { x:a[0], y:a[1] };
      pts.push({ ...cur });
    }

    else if (cmd === 'C') {
      const p0 = { ...cur };
      const p1 = { x:a[0], y:a[1] };
      const p2 = { x:a[2], y:a[3] };
      const p3 = { x:a[4], y:a[5] };
      const sampled = adaptiveSampleBezier(p0,p1,p2,p3,maxSeg,14);
      for (let i=1;i<sampled.length;i++) pts.push(sampled[i]);
      cur = p3;
    }

    else if (cmd === 'Q') {
      const p0 = { ...cur };
      const p1 = { x:a[0], y:a[1] };
      const p2 = { x:a[2], y:a[3] };
      // convert Q → cubic
      const c1 = { x: p0.x + 2/3*(p1.x-p0.x), y: p0.y + 2/3*(p1.y-p0.y) };
      const c2 = { x: p2.x + 2/3*(p1.x-p2.x), y: p2.y + 2/3*(p1.y-p2.y) };
      const sampled = adaptiveSampleBezier(p0,c1,c2,p2,maxSeg,14);
      for (let i=1;i<sampled.length;i++) pts.push(sampled[i]);
      cur = p2;
    }

    else if (cmd === 'A') {
      // approximate arc by endpoints (simple fallback)
      for (let i=0;i<a.length;i+=7) {
        const x = a[i+5], y = a[i+6];
        cur = { x, y };
        pts.push({ ...cur });
      }
    }
  }

  return pts;
}

// ---------------------------------------------------------------
//  OFFSET POLYGON (clipper-lib)
// ---------------------------------------------------------------
function offsetPolygon(poly: Polyline, offset: number): Polyline {
  const scale = 1000;
  const subj = poly.map(p => ({ X: Math.round(p.x*scale), Y: Math.round(p.y*scale) }));
  const co = new ClipperLib.ClipperOffset();
  co.AddPath(subj, ClipperLib.JoinType.jtRound, ClipperLib.EndType.etClosedPolygon);
  const solution = new ClipperLib.Paths();
  co.Execute(solution, offset*scale);
  if (!solution.length) return [];
  return solution[0].map(pt => ({ x:pt.X/scale, y:pt.Y/scale }));
}

// ---------------------------------------------------------------
//  G2 CONTINUITY SOLVER (smooth join between curves)
// ---------------------------------------------------------------
function solveG2(p0:Point,p1:Point,p2:Point,p3:Point, q0:Point,q1:Point,q2:Point,q3:Point) {
  // simplified version — adjusts p2 and q1
  let P2 = { ...p2 }, Q1 = { ...q1 };
  const step = 0.2;
  for (let iter=0; iter<60; iter++) {
    const d1A = deriv1(p0,p1,P2,p3,1);
    const d1B = deriv1(q0,Q1,q2,q3,0);
    const d2A = deriv2(p0,p1,P2,p3,1);
    const d2B = deriv2(q0,Q1,q2,q3,0);

    const r1 = { x:d1A.x-d1B.x, y:d1A.y-d1B.y };
    const r2 = { x:d2A.x-d2B.x, y:d2A.y-d2B.y };

    const cost = r1.x*r1.x+r1.y*r1.y + 0.6*(r2.x*r2.x+r2.y*r2.y);
    if (cost < 1e-6) break;

    // gradient approx
    P2.x -= step*r1.x*0.01;
    P2.y -= step*r1.y*0.01;
    Q1.x += step*r1.x*0.01;
    Q1.y += step*r1.y*0.01;
  }
  return { p2:P2, q1:Q1 };

  function deriv1(a:Point,b:Point,c:Point,d:Point,t:number) {
    const u = 1-t;
    return {
      x: 3*u*u*(b.x-a.x) + 6*u*t*(c.x-b.x) + 3*t*t*(d.x-c.x),
      y: 3*u*u*(b.y-a.y) + 6*u*t*(c.y-b.y) + 3*t*t*(d.y-c.y)
    };
  }
  function deriv2(a:Point,b:Point,c:Point,d:Point,t:number) {
    const u = 1-t;
    return {
      x: 6*u*(c.x-2*b.x+a.x) + 6*t*(d.x-2*c.x+b.x),
      y: 6*u*(c.y-2*b.y+a.y) + 6*t*(d.y-2*c.y+b.y)
    };
  }
}
