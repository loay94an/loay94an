private _createMeasurementGuide(
    x1: number, y1: number,
    x2: number, y2: number,
    offset: number,
    label: string,
    unit: string = 'سم'
): SvgPath[] {
    const scale = this.scale;
    const paths: SvgPath[] = [];
    
    // خط القياس الرئيسي
    const mainLine = `M ${x1 * scale} ${y1 * scale} L ${x2 * scale} ${y2 * scale}`;
    paths.push({ d: mainLine, class: 'stroke-gray-400 stroke-1 stroke-dasharray' });
    
    // إضافة علامات البداية والنهاية
    const arrowSize = 0.3 * scale;
    paths.push({
        d: `M ${x1 * scale - arrowSize} ${y1 * scale - arrowSize} 
            L ${x1 * scale} ${y1 * scale} 
            L ${x1 * scale + arrowSize} ${y1 * scale - arrowSize}`,
        class: 'stroke-gray-600 stroke-1'
    });
    
    paths.push({
        d: `M ${x2 * scale - arrowSize} ${y2 * scale - arrowSize} 
            L ${x2 * scale} ${y2 * scale} 
            L ${x2 * scale + arrowSize} ${y2 * scale - arrowSize}`,
        class: 'stroke-gray-600 stroke-1'
    });
    
    // نص القياس
    const midX = (x1 + x2) / 2;
    const midY = (y1 + y2) / 2;
    
    paths.push({
        d: '',
        label: `${label}: ${Math.sqrt(Math.pow(x2-x1,2) + Math.pow(y2-y1,2)).toFixed(1)} ${unit}`,
        class: 'fill-gray-700 text-xs font-medium',
        x: midX * scale + offset,
        y: midY * scale
    });
    
    return paths;
}
