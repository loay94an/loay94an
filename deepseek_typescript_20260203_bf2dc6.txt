generatePatternForPrint(
    model: PatternModel, 
    measurements: Measurements, 
    options: {
        paperSize?: 'A4' | 'A3' | 'A2',
        scale?: number,
        includeInstructions?: boolean
    } = {}
): PatternData {
    const pattern = this.generatePattern(model, measurements);
    
    // تطبيق حجم الورقة
    if (options.paperSize) {
        const paperSizes = {
            'A4': { width: 210, height: 297 },
            'A3': { width: 297, height: 420 },
            'A2': { width: 420, height: 594 }
        };
        
        const paper = paperSizes[options.paperSize];
        const scaleFactor = Math.min(
            paper.width / pattern.width,
            paper.height / pattern.height
        ) * 0.9; // هامش 10%
        
        // تطبيق عامل التحجيم
        pattern.paths = pattern.paths.map(path => ({
            ...path,
            d: path.d?.replace(/(\d+(\.\d+)?)/g, match => 
                (parseFloat(match) * scaleFactor).toString()
            ),
            x: path.x ? path.x * scaleFactor : undefined,
            y: path.y ? path.y * scaleFactor : undefined
        }));
        
        pattern.width *= scaleFactor;
        pattern.height *= scaleFactor;
    }
    
    // إضافة تعليمات الطباعة إذا لزم الأمر
    if (options.includeInstructions) {
        this._addPrintingInstructions(pattern.paths, model, measurements);
    }
    
    return pattern;
}

private _addPrintingInstructions(paths: SvgPath[], model: PatternModel, measurements: Measurements): void {
    paths.push({
        d: '',
        label: `بترون ${model.name} - ${new Date().toLocaleDateString('ar-SA')}`,
        class: 'fill-gray-800 text-lg font-bold',
        x: 10,
        y: 20
    });
    
    paths.push({
        d: '',
        label: `المقاس: ${measurements.bust}/${measurements.waist}/${measurements.hips}`,
        class: 'fill-gray-600 text-sm',
        x: 10,
        y: 40
    });
    
    paths.push({
        d: '',
        label: 'قص 1 على الثنية ما لم يذكر غير ذلك',
        class: 'fill-blue-600 text-sm italic',
        x: 10,
        y: 60
    });
}
