// ---------------------------------------------------------------
//  BASIC TYPES
// ---------------------------------------------------------------
export interface Point { x: number; y: number; }
export interface Polyline extends Array<Point> {}
export interface SvgPath { d?: string; class?: string; label?: string; x?: number; y?: number; }

// ---------------------------------------------------------------
//  PATH BUILDER (clean SVG generator)
// ---------------------------------------------------------------
class PathBuilder {
  private d: string[] = [];

  moveTo(x: number, y: number) { this.d.push(`M ${x} ${y}`); return this; }
  lineTo(x: number, y: number) { this.d.push(`L ${x} ${y}`); return this; }
  cubic(x1:number,y1:number,x2:number,y2:number,x:number,y:number) {
    this.d.push(`C ${x1} ${y1}, ${x2} ${y2}, ${x} ${y}`); return this;
  }
  quad(x1:number,y1:number,x:number,y:number) {
    this.d.push(`Q ${x1} ${y1}, ${x} ${y}`); return this;
  }
  arc(rx:number,ry:number,rot:number,laf:number,sf:number,x:number,y:number) {
    this.d.push(`A ${rx} ${ry} ${rot} ${laf} ${sf} ${x} ${y}`); return this;
  }
  close() { this.d.push('Z'); return this; }
  build() { return this.d.join(' '); }
}

// ---------------------------------------------------------------
//  ADAPTIVE BEZIER SAMPLING (de Casteljau)
// ---------------------------------------------------------------
function adaptiveSampleBezier(
  p0:Point, p1:Point, p2:Point, p3:Point,
  maxSeg = 0.5, maxDepth = 14
): Point[] {

  const out: Point[] = [];
  const toArr = (p:Point) => [p.x, p.y] as [number,number];

  const bez = (a:[number,number],b:[number,number],c:[number,number],d:[number,number],t:number) => {
    const u = 1 - t;
    return {
      x: u*u*u*a[0] + 3*u*u*t*b[0] + 3*u*t*t*c[0] + t*t*t*d[0],
      y: u*u*u*a[1] + 3*u*u*t*b[1] + 3*u*t*t*c[1] + t*t*t*d[1]
    };
  };

  const subdiv = (a:[number,number],b:[number,number],c:[number,number],d:[number,number],depth:number) => {
    const mid = bez(a,b,c,d,0.5);
    const lm = [(a[0]+d[0])/2,(a[1]+d[1])/2];
    const dx = mid.x - lm[0], dy = mid.y - lm[1];

    if (Math.hypot(dx,dy) > maxSeg && depth < maxDepth) {
      const a1:[number,number] = [(a[0]+b[0])/2,(a[1]+b[1])/2];
      const b1:[number,number] = [(b[0]+c[0])/2,(b[1]+c[1])/2];
      const c1:[number,number] = [(c[0]+d[0])/2,(c[1]+d[1])/2];
      const a2:[number,number] = [(a1[0]+b1[0])/2,(a1[1]+b1[1])/2];
      const b2:[number,number] = [(b1[0]+c1[0])/2,(b1[1]+c1[1])/2];
      const m:[number,number] = [(a2[0]+b2[0])/2,(a2[1]+b2[1])/2];

      subdiv(a,a1,a2,m,depth+1);
      subdiv(m,b2,c1,d,depth+1);
    } else {
      out.push({ x:a[0], y:a[1] });
      out.push({ x:d[0], y:d[1] });
    }
  };

  subdiv(toArr(p0),toArr(p1),toArr(p2),toArr(p3),0);

  // remove duplicates
  const res: Point[] = [];
  for (const pt of out)
    if (!res.length || Math.hypot(res[res.length-1].x-pt.x,res[res.length-1].y-pt.y)>1e-9)
      res.push(pt);

  return res;
}

// ---------------------------------------------------------------
//  BEZIER LENGTH (via adaptive sampling)
// ---------------------------------------------------------------
function bezierLength(p0:Point,p1:Point,p2:Point,p3:Point): number {
  const pts = adaptiveSampleBezier(p0,p1,p2,p3,0.5,14);
  let L = 0;
  for (let i=1;i<pts.length;i++)
    L += Math.hypot(pts[i].x-pts[i-1].x, pts[i].y-pts[i-1].y);
  return L;
}

// ---------------------------------------------------------------
//  SVG PATH PARSER (supports M,L,C,Q,A,Z)
// ---------------------------------------------------------------
function parsePath(d: string) {
  const out: { cmd:string; args:number[] }[] = [];
  const re = /([a-zA-Z])([^a-zA-Z]*)/g;
  let m;
  while ((m = re.exec(d)) !== null) {
    const cmd = m[1].toUpperCase();
    const nums = m[2].trim()
      ? m[2].trim().split(/[\s,]+/).map(Number).filter(n=>!isNaN(n))
      : [];
    out.push({ cmd, args: nums });
  }
  return out;
}

// ---------------------------------------------------------------
//  DENSIFY PATH (convert curves → polyline)
// ---------------------------------------------------------------
function densifyPath(d: string, maxSeg = 0.5): Polyline {
  const tokens = parsePath(d);
  const pts: Polyline = [];
  let cur: Point = { x:0, y:0 };

  for (const t of tokens) {
    const cmd = t.cmd;
    const a = t.args;

    if (cmd === 'M') {
      cur = { x:a[0], y:a[1] };
      pts.push({ ...cur });
    }

    else if (cmd === 'L') {
      cur = { x:a[0], y:a[1] };
      pts.push({ ...cur });
    }

    else if (cmd === 'C') {
      const p0 = { ...cur };
      const p1 = { x:a[0], y:a[1] };
      const p2 = { x:a[2], y:a[3] };
      const p3 = { x:a[4], y:a[5] };
      const sampled = adaptiveSampleBezier(p0,p1,p2,p3,maxSeg,14);
      for (let i=1;i<sampled.length;i++) pts.push(sampled[i]);
      cur = p3;
    }

    else if (cmd === 'Q') {
      const p0 = { ...cur };
      const p1 = { x:a[0], y:a[1] };
      const p2 = { x:a[2], y:a[3] };
      // convert Q → cubic
      const c1 = { x: p0.x + 2/3*(p1.x-p0.x), y: p0.y + 2/3*(p1.y-p0.y) };
      const c2 = { x: p2.x + 2/3*(p1.x-p2.x), y: p2.y + 2/3*(p1.y-p2.y) };
      const sampled = adaptiveSampleBezier(p0,c1,c2,p2,maxSeg,14);
      for (let i=1;i<sampled.length;i++) pts.push(sampled[i]);
      cur = p2;
    }

    else if (cmd === 'A') {
      // approximate arc by endpoints (simple fallback)
      for (let i=0;i<a.length;i+=7) {
        const x = a[i+5], y = a[i+6];
        cur = { x, y };
        pts.push({ ...cur });
      }
    }
  }

  return pts;
}

// ---------------------------------------------------------------
//  OFFSET POLYGON (clipper-lib)
// ---------------------------------------------------------------
function offsetPolygon(poly: Polyline, offset: number): Polyline {
  const scale = 1000;
  const subj = poly.map(p => ({ X: Math.round(p.x*scale), Y: Math.round(p.y*scale) }));
  const co = new ClipperLib.ClipperOffset();
  co.AddPath(subj, ClipperLib.JoinType.jtRound, ClipperLib.EndType.etClosedPolygon);
  const solution = new ClipperLib.Paths();
  co.Execute(solution, offset*scale);
  if (!solution.length) return [];
  return solution[0].map(pt => ({ x:pt.X/scale, y:pt.Y/scale }));
}

// ---------------------------------------------------------------
//  G2 CONTINUITY SOLVER (smooth join between curves)
// ---------------------------------------------------------------
function solveG2(p0:Point,p1:Point,p2:Point,p3:Point, q0:Point,q1:Point,q2:Point,q3:Point) {
  // simplified version — adjusts p2 and q1
  let P2 = { ...p2 }, Q1 = { ...q1 };
  const step = 0.2;
  for (let iter=0; iter<60; iter++) {
    const d1A = deriv1(p0,p1,P2,p3,1);
    const d1B = deriv1(q0,Q1,q2,q3,0);
    const d2A = deriv2(p0,p1,P2,p3,1);
    const d2B = deriv2(q0,Q1,q2,q3,0);

    const r1 = { x:d1A.x-d1B.x, y:d1A.y-d1B.y };
    const r2 = { x:d2A.x-d2B.x, y:d2A.y-d2B.y };

    const cost = r1.x*r1.x+r1.y*r1.y + 0.6*(r2.x*r2.x+r2.y*r2.y);
    if (cost < 1e-6) break;

    // gradient approx
    P2.x -= step*r1.x*0.01;
    P2.y -= step*r1.y*0.01;
    Q1.x += step*r1.x*0.01;
    Q1.y += step*r1.y*0.01;
  }
  return { p2:P2, q1:Q1 };

  function deriv1(a:Point,b:Point,c:Point,d:Point,t:number) {
    const u = 1-t;
    return {
      x: 3*u*u*(b.x-a.x) + 6*u*t*(c.x-b.x) + 3*t*t*(d.x-c.x),
      y: 3*u*u*(b.y-a.y) + 6*u*t*(c.y-b.y) + 3*t*t*(d.y-c.y)
    };
  }
  function deriv2(a:Point,b:Point,c:Point,d:Point,t:number) {
    const u = 1-t;
    return {
      x: 6*u*(c.x-2*b.x+a.x) + 6*t*(d.x-2*c.x+b.x),
      y: 6*u*(c.y-2*b.y+a.y) + 6*t*(d.y-2*c.y+b.y)
    };
  }
}

@Injectable({ providedIn: 'root' })
export class PatternServiceCore {
  PathBuilder = PathBuilder;
  parsePath = parsePath;
  densifyPath = densifyPath;
  adaptiveSampleBezier = adaptiveSampleBezier;
  bezierLength = bezierLength;
  offsetPolygon = offsetPolygon;
  solveG2 = solveG2;
}
// ===============================================================
//  PATTERN SERVICE — ULTRA PRO EDITION (PART 2 — GENERATORS)
//  High‑precision pattern generators (bodice, sleeve, pants, skirt…)
// ===============================================================

import { Injectable } from '@angular/core';
import { PatternServiceCore, Point, Polyline, SvgPath } from './pattern-core.service';

// ---------------------------------------------------------------
//  MAIN GENERATOR SERVICE (depends on CORE ENGINE)
// ---------------------------------------------------------------
@Injectable({ providedIn: 'root' })
export class PatternGenerators {

  constructor(private core: PatternServiceCore) {}

  // -------------------------------------------------------------
  //  BODICE GENERATOR (Ultra‑Pro)
  // -------------------------------------------------------------
  generateBodice(meas: any, isFront: boolean, category: string) {

    const scale = 10;

    // Extract measurements
    const bust = meas.bust ?? 92;
    const waist = meas.waist ?? 74;
    const hips = meas.hips ?? 98;
    const shoulder = meas.shoulderWidth ?? 14;
    const length = meas.pieceLength ?? 42;

    // Ease
    const bustEase = category === 'man' ? 2.5 : 2;
    const waistEase = 2.5;
    const hipsEase = category === 'man' ? 2 : 1.5;

    const QB = bust/4 + bustEase;
    const QW = waist/4 + waistEase;
    const QH = hips/4 + hipsEase;

    const armDepth = (bust/10) + (category==='woman'?10.5:12);
    const neckWidth = (bust/20)+2;
    const neckDrop = isFront ? (bust/12)+1 : 2;
    const shoulderSlope = category==='child'?3:4.5;

    const chestWidth = (bust/4)-4;
    const backWidth = (bust/8)+5.5;

    // Key points
    const p = {
      neckTop: { x: neckWidth, y: 0 },
      neckDrop: { x: 0, y: neckDrop },
      shoulderTip: { x: shoulder/2, y: shoulderSlope },
      bustSide: { x: QB, y: armDepth },
      waistSide: { x: QW, y: 42 },
      hipSide: { x: QH, y: 62 },
      hemSide: { x: QH, y: length },
      armLand: { x: isFront? chestWidth : backWidth, y: armDepth }
    };

    // Neckline control points
    const nC1 = { x: p.neckDrop.x + p.neckTop.x*0.55, y: p.neckDrop.y };
    const nC2 = { x: p.neckTop.x*0.9, y: p.neckDrop.y*0.55 };

    // Armhole control points
    const h = p.bustSide.y - p.shoulderTip.y;
    const aC1 = {
      x: p.shoulderTip.x - (isFront?2.5:0.75),
      y: p.shoulderTip.y + h*0.35
    };
    const aC2 = {
      x: isFront
        ? p.armLand.x + (p.bustSide.x - p.armLand.x)*0.2
        : p.armLand.x + (p.bustSide.x - p.armLand.x)*0.7,
      y: p.bustSide.y - h*0.30
    };

    // Build path
    const PB = new this.core.PathBuilder()
      .moveTo(0*scale, p.hemSide.y*scale)
      .lineTo(0*scale, p.waistSide.y*scale)
      .lineTo(p.neckDrop.x*scale, p.neckDrop.y*scale)
      .cubic(nC1.x*scale,nC1.y*scale,nC2.x*scale,nC2.y*scale,p.neckTop.x*scale,p.neckTop.y*scale)
      .lineTo(p.shoulderTip.x*scale,p.shoulderTip.y*scale)
      .cubic(aC1.x*scale,aC1.y*scale,aC2.x*scale,aC2.y*scale,p.bustSide.x*scale,p.bustSide.y*scale)
      .lineTo(p.hemSide.x*scale,p.hemSide.y*scale)
      .close();

    const d = PB.build();

    // Compute armhole length
    const armLen = this.core.bezierLength(
      p.shoulderTip, aC1, aC2, p.bustSide
    );

    // Notches (sampling-based)
    const sampled = this.core.adaptiveSampleBezier(
      p.shoulderTip, aC1, aC2, p.bustSide, 0.5, 14
    );
    const notch = (r:number) => {
      const i = Math.floor(r*(sampled.length-1));
      const pt = sampled[i];
      const L = 0.5*scale;
      return `M ${pt.x*scale-L} ${pt.y*scale} L ${pt.x*scale+L} ${pt.y*scale}`;
    };

    const paths: SvgPath[] = [
      { d, class:'stroke-black fill-none' }
    ];

    if (isFront) {
      paths.push({ d: notch(0.65), class:'stroke-red-600' });
    } else {
      paths.push({ d: notch(0.60), class:'stroke-red-600' });
      paths.push({ d: notch(0.65), class:'stroke-red-600' });
    }

    return {
      paths,
      width: QB*scale + 10*scale,
      height: length*scale,
      armhole: armLen
    };
  }

  // -------------------------------------------------------------
  //  SLEEVE GENERATOR (Ultra‑Pro)
  // -------------------------------------------------------------
  generateSleeve(meas:any, armLen:number, type:string) {

    const scale = 10;
    const L = meas.sleeveLength ?? 60;
    const AC = meas.armCircumference ?? 28;

    const bicep = AC + 6;
    const half = bicep/2;
    const capH = armLen/3;

    const p = {
      capTop: { x:half, y:0 },
      left: { x:0, y:capH },
      right: { x:bicep, y:capH },
      hemL: { x:half-10, y:L },
      hemR: { x:half+10, y:L }
    };

    const cap = {
      b1: { x:p.left.x + half*0.3, y:p.left.y - capH*1.1 },
      b2: { x:p.capTop.x - half*0.6, y:p.capTop.y },
      f1: { x:p.capTop.x + half*0.7, y:p.capTop.y },
      f2: { x:p.right.x - half*0.15, y:p.right.y - capH*0.9 }
    };

    const PB = new this.core.PathBuilder()
      .moveTo(p.hemL.x*scale,p.hemL.y*scale)
      .lineTo(p.left.x*scale,p.left.y*scale)
      .cubic(cap.b1.x*scale,cap.b1.y*scale,cap.b2.x*scale,cap.b2.y*scale,p.capTop.x*scale,p.capTop.y*scale)
      .cubic(cap.f1.x*scale,cap.f1.y*scale,cap.f2.x*scale,cap.f2.y*scale,p.right.x*scale,p.right.y*scale)
      .lineTo(p.hemR.x*scale,p.hemR.y*scale)
      .close();

    const d = PB.build();

    // Notches
    const frontSample = this.core.adaptiveSampleBezier(p.capTop,cap.f1,cap.f2,p.right,0.5,14);
    const backSample  = this.core.adaptiveSampleBezier(p.left,cap.b1,cap.b2,p.capTop,0.5,14);

    const pick = (arr:Point[], r:number) => {
      const i = Math.floor(r*(arr.length-1));
      const pt = arr[i];
      const L = 0.5*scale;
      return `M ${pt.x*scale-L} ${pt.y*scale} L ${pt.x*scale+L} ${pt.y*scale}`;
    };

    const paths: SvgPath[] = [
      { d, class:'stroke-black fill-none' },
      { d: pick(frontSample,0), class:'stroke-red-600' },
      { d: pick(frontSample,0.65), class:'stroke-red-600' },
      { d: pick(backSample,1-0.60), class:'stroke-red-600' },
      { d: pick(backSample,1-0.65), class:'stroke-red-600' }
    ];

    return {
      paths,
      width: (p.right.x-p.left.x)*scale + 20*scale,
      height: L*scale
    };
  }

  // -------------------------------------------------------------
  //  PANTS GENERATOR (simplified Ultra‑Pro)
  // -------------------------------------------------------------
  generatePants(meas:any, isFront:boolean) {
    const scale = 10;
    const waist = meas.waist ?? 74;
    const hips = meas.hips ?? 98;
    const length = meas.pantsLength ?? 100;

    const QW = waist/4 + 2;
    const QH = hips/4 + 2;

    const PB = new this.core.PathBuilder()
      .moveTo(0,0)
      .lineTo(QW*scale,0)
      .lineTo(QH*scale,length*scale)
      .lineTo(0,length*scale)
      .close();

    return {
      paths: [{ d:PB.build(), class:'stroke-black fill-none' }],
      width: QH*scale + 10*scale,
      height: length*scale
    };
  }

  // -------------------------------------------------------------
  //  SKIRT GENERATOR (simplified Ultra‑Pro)
  // -------------------------------------------------------------
  generateSkirt(meas:any) {
    const scale = 10;
    const waist = meas.waist ?? 74;
    const hips = meas.hips ?? 98;
    const length = meas.skirtLength ?? 60;

    const QW = waist/4 + 2;
    const QH = hips/4 + 2;

    const PB = new this.core.PathBuilder()
      .moveTo(0,0)
      .lineTo(QW*scale,0)
      .lineTo(QH*scale,length*scale)
      .lineTo(0,length*scale)
      .close();

    return {
      paths: [{ d:PB.build(), class:'stroke-black fill-none' }],
      width: QH*scale + 10*scale,
      height: length*scale
    };
  }

  // -------------------------------------------------------------
  //  COLLAR GENERATOR (simple)
  // -------------------------------------------------------------
  generateCollar(meas:any) {
    const scale = 10;
    const neck = meas.neckCircumference ?? 38;

    const PB = new this.core.PathBuilder()
      .moveTo(0,0)
      .lineTo(neck*scale/2,0)
      .lineTo(neck*scale/2,30)
      .lineTo(0,30)
      .close();

    return {
      paths: [{ d:PB.build(), class:'stroke-black fill-none' }],
      width: (neck/2)*scale + 20,
      height: 40
    };
  }

  // -------------------------------------------------------------
  //  PLACEHOLDERS FOR IMAGE‑BASED PATTERNS
  // -------------------------------------------------------------
  generateFromImage(id:string) {
    return [{
      paths: [{ d:'M 0 0 L 100 0 L 100 200 L 0 200 Z', class:'stroke-blue-600 fill-none' }],
      width: 100,
      height: 200
    }];
  }
}
// ===============================================================
//  PATTERN SERVICE — ULTRA PRO EDITION (PART 3 — FACTORY + ASSEMBLY)
//  Final assembly of pattern pieces into PatternData
// ===============================================================

import { Injectable } from '@angular/core';
import { PatternServiceCore, SvgPath, Point, Polyline } from './pattern-core.service';
import { PatternGenerators } from './pattern-generators.service';

export interface PatternData {
  paths: SvgPath[];
  width: number;
  height: number;
}

@Injectable({ providedIn: 'root' })
export class PatternService {

  constructor(
    private core: PatternServiceCore,
    private gen: PatternGenerators
  ) {}

  // -------------------------------------------------------------
  //  MAIN ENTRY POINT
  // -------------------------------------------------------------
  generatePattern(model: any, meas: any): PatternData {

    const all: SvgPath[] = [];
    let xOffset = 0;
    let maxHeight = 0;
    const pad = 80; // spacing between pieces

    // -----------------------------------------------------------
    // 1) BODICE (front + back)
    // -----------------------------------------------------------
    if (model.includes('bodice') || model.includes('shirt') || model.includes('dress')) {

      const front = this.gen.generateBodice(meas, true, 'woman');
      this.offsetPaths(front.paths, xOffset, 0);
      all.push(...front.paths);
      xOffset += front.width + pad;
      maxHeight = Math.max(maxHeight, front.height);

      const back = this.gen.generateBodice(meas, false, 'woman');
      this.offsetPaths(back.paths, xOffset, 0);
      all.push(...back.paths);
      xOffset += back.width + pad;
      maxHeight = Math.max(maxHeight, back.height);

      // sleeve
      const sleeve = this.gen.generateSleeve(meas, front.armhole + back.armhole, 'basic');
      this.offsetPaths(sleeve.paths, xOffset, 0);
      all.push(...sleeve.paths);
      xOffset += sleeve.width + pad;
      maxHeight = Math.max(maxHeight, sleeve.height);
    }

    // -----------------------------------------------------------
    // 2) PANTS
    // -----------------------------------------------------------
    if (model.includes('pants')) {
      const front = this.gen.generatePants(meas, true);
      this.offsetPaths(front.paths, xOffset, 0);
      all.push(...front.paths);
      xOffset += front.width + pad;
      maxHeight = Math.max(maxHeight, front.height);

      const back = this.gen.generatePants(meas, false);
      this.offsetPaths(back.paths, xOffset, 0);
      all.push(...back.paths);
      xOffset += back.width + pad;
      maxHeight = Math.max(maxHeight, back.height);
    }

    // -----------------------------------------------------------
    // 3) SKIRT
    // -----------------------------------------------------------
    if (model.includes('skirt')) {
      const skirt = this.gen.generateSkirt(meas);
      this.offsetPaths(skirt.paths, xOffset, 0);
      all.push(...skirt.paths);
      xOffset += skirt.width + pad;
      maxHeight = Math.max(maxHeight, skirt.height);
    }

    // -----------------------------------------------------------
    // 4) COLLAR
    // -----------------------------------------------------------
    if (model.includes('collar')) {
      const col = this.gen.generateCollar(meas);
      this.offsetPaths(col.paths, xOffset, 0);
      all.push(...col.paths);
      xOffset += col.width + pad;
      maxHeight = Math.max(maxHeight, col.height);
    }

    // -----------------------------------------------------------
    // 5) IMAGE‑BASED PATTERNS
    // -----------------------------------------------------------
    if (model.includes('image')) {
      const pieces = this.gen.generateFromImage(model);
      for (const p of pieces) {
        this.offsetPaths(p.paths, xOffset, 0);
        all.push(...p.paths);
        xOffset += p.width + pad;
        maxHeight = Math.max(maxHeight, p.height);
      }
    }

    // -----------------------------------------------------------
    // 6) FINAL RETURN
    // -----------------------------------------------------------
    return {
      paths: all,
      width: xOffset,
      height: maxHeight + pad
    };
  }

  // -------------------------------------------------------------
  //  OFFSET PATHS (translate SVG paths)
  // -------------------------------------------------------------
  private offsetPaths(paths: SvgPath[], dx: number, dy: number) {
    for (const p of paths) {
      if (p.d) {
        const tokens = this.core.parsePath(p.d);
        const moved = tokens.map(t => {
          const a = t.args.slice();
          for (let i=0;i<a.length;i+=2) {
            a[i] += dx;
            a[i+1] += dy;
          }
          return { cmd:t.cmd, args:a };
        });
        p.d = moved.map(t => `${t.cmd} ${t.args.join(' ')}`).join(' ');
      }
      if (p.x !== undefined) p.x += dx;
      if (p.y !== undefined) p.y += dy;
    }
  }
}
