private _generateSeamAllowance(pathPoints: {x: number, y: number}[], offset: number, closed: boolean = true): string {
    if (pathPoints.length < 2) return '';
    
    const builder = new PathBuilder();
    const offsetPoints = this._calculateOffsetPath(pathPoints, offset, closed);
    
    if (offsetPoints.length > 0) {
        builder.moveTo(offsetPoints[0].x * this.scale, offsetPoints[0].y * this.scale);
        
        for (let i = 1; i < offsetPoints.length; i++) {
            // استخدام منحنيات بيزير للزوايا بدلاً من الخطوط المستقيمة
            if (i % 3 === 0 && i < offsetPoints.length - 3) {
                const cp1 = {
                    x: (offsetPoints[i-1].x + offsetPoints[i].x) / 2,
                    y: offsetPoints[i-1].y
                };
                const cp2 = {
                    x: offsetPoints[i].x,
                    y: (offsetPoints[i-1].y + offsetPoints[i].y) / 2
                };
                builder.cubicCurveTo(
                    cp1.x * this.scale, cp1.y * this.scale,
                    cp2.x * this.scale, cp2.y * this.scale,
                    offsetPoints[i].x * this.scale, offsetPoints[i].y * this.scale
                );
            } else {
                builder.lineTo(offsetPoints[i].x * this.scale, offsetPoints[i].y * this.scale);
            }
        }
        
        if (closed) builder.closePath();
    }
    
    return builder.build();
}
